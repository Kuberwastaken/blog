719	def preserve_enrichment_tables(conn: sqlite3.Connection, db_path: Path) -> None:
720	    """Copy enrichment/gemini rows from the existing DB into the freshly built
721	    (tmp) DB so rebuilding ``leads`` does not wipe enrichment work. Uses only the
722	    columns shared by both schemas, and lets SQLite bump ``sqlite_sequence`` for
723	    AUTOINCREMENT tables from the explicit ids we re-insert.
724	    """
725	    if not db_path.exists():
726	        return
727	    conn.execute("ATTACH DATABASE ? AS old", (str(db_path),))
728	    try:
729	        old_tables = {
730	            row[0]
731	            for row in conn.execute(
732	                "SELECT name FROM old.sqlite_master WHERE type='table'"
733	            ).fetchall()
734	        }
735	        preserved: dict[str, int] = {}
736	        for table in ENRICHMENT_TABLES:
737	            if table not in old_tables:
738	                continue
739	            new_cols = [r[1] for r in conn.execute(f"PRAGMA main.table_info({table})").fetchall()]
740	            if not new_cols:
741	                continue
742	            old_cols = {r[1] for r in conn.execute(f"PRAGMA old.table_info({table})").fetchall()}
743	            shared = [c for c in new_cols if c in old_cols]
744	            if not shared:
745	                continue
746	            col_list = ",".join(shared)
747	            conn.execute(f"INSERT INTO main.{table} ({col_list}) SELECT {col_list} FROM old.{table}")
748	            preserved[table] = conn.execute(f"SELECT COUNT(*) FROM main.{table}").fetchone()[0]
749	        conn.commit()
750	        if preserved:
751	            print("preserved enrichment tables: " + ", ".join(f"{k}={v:,}" for k, v in preserved.items()))
752	    finally:
753	        conn.execute("DETACH DATABASE old")

767	def where_from_params(params: dict[str, list[str]]) -> tuple[str, list[Any]]:
768	    clauses: list[str] = []
769	    values: list[Any] = []
770	    q = clean(params.get("q", [""])[0])
771	    if q:
772	        clauses.append("l.search_text LIKE ?")
773	        values.append(f"%{q.lower()}%")
774	
775	    for param, column in MULTI_FILTERS.items():
776	        vals = split_values(params, param)
777	        if vals:
778	            clauses.append(f"{column} IN ({','.join(['?'] * len(vals))})")
779	            values.extend(vals)
780	
781	    signals = split_values(params, "signals")
782	    for signal in signals:
783	        if signal in BOOL_FILTERS:
784	            clauses.append(f"l.{signal}=1")
785	
786	    for signal in BOOL_FILTERS:
787	        raw = params.get(signal, [""])[0]
788	        if raw in {"1", "0"}:
789	            clauses.append(f"l.{signal}=?")
790	            values.append(int(raw))
791	
792	    route_vals = split_values(params, "enrichment_route")
793	    if route_vals:
794	        clauses.append(f"{ENRICHMENT_ROUTE_CASE} IN ({','.join(['?'] * len(route_vals))})")
795	        values.extend(route_vals)
796	
797	    status_vals = split_values(params, "enrichment_status")
798	    if status_vals:
799	        clauses.append(f"{ENRICHMENT_STATUS_EXPR} IN ({','.join(['?'] * len(status_vals))})")
800	        values.extend(status_vals)
801	
802	    website_filter = clean(params.get("website_filter", [""])[0])
803	    if website_filter == "has":
804	        clauses.append("l.has_domain=1")
805	    elif website_filter == "missing":
806	        clauses.append("l.has_domain=0")
807	
808	    email_filter = clean(params.get("email_filter", [""])[0])
809	    if email_filter == "has":
810	        clauses.append("l.has_email=1")
811	    elif email_filter == "missing":
812	        clauses.append("l.has_email=0")
813	    elif email_filter == "corporate":
814	        clauses.append("l.corporate_email=1")
815	    elif email_filter == "free":
816	        clauses.append("l.free_email=1")
817	
818	    phone_filter = clean(params.get("phone_filter", [""])[0])
819	    if phone_filter == "has":
820	        clauses.append("l.has_phone=1")
821	    elif phone_filter == "missing":
822	        clauses.append("l.has_phone=0")
823	
824	    registry_filter = clean(params.get("registry_filter", [""])[0])
825	    if registry_filter == "any":
826	        clauses.append("(l.has_cin=1 OR l.has_gstin=1 OR l.active_registry=1)")
827	    elif registry_filter == "none":
